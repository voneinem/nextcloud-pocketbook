# Qt layer between the QML UI and pbc_core. ARM: SDK Qt with manual moc. HOST: host Qt6 (Core only)
# when installed; otherwise the bridge, the app and the bridge tests are simply not built.

set(PBC_BRIDGE_SOURCES src/about_bridge.cpp src/worker.cpp src/settings_bridge.cpp src/table_translator.cpp)
set(PBC_BRIDGE_HEADERS include/about_bridge.h include/worker.h include/settings_bridge.h include/table_translator.h)
# headers with Q_OBJECT (the device build runs moc by hand on these)
set(PBC_BRIDGE_MOC_HEADERS include/about_bridge.h include/settings_bridge.h)

if(TARGET_TYPE STREQUAL "ARM")
    pbc_moc(PBC_BRIDGE_MOC ${PBC_BRIDGE_MOC_HEADERS})
    add_library(pbc_bridge STATIC ${PBC_BRIDGE_SOURCES} ${PBC_BRIDGE_MOC})
    target_include_directories(pbc_bridge PUBLIC include)
    target_include_directories(pbc_bridge SYSTEM PUBLIC ${PBC_QT_INCLUDES})
    target_compile_definitions(pbc_bridge PUBLIC PLATFORM_POCKETBOOK)
    target_link_libraries(pbc_bridge PUBLIC pbc_core ${PBC_QT_LIBS})
else()
    find_package(Qt6 QUIET COMPONENTS Core Gui Quick)
    if(Qt6Core_FOUND AND Qt6Gui_FOUND AND Qt6Quick_FOUND)
        set(CMAKE_AUTOMOC ON)
        add_library(pbc_bridge STATIC ${PBC_BRIDGE_SOURCES} ${PBC_BRIDGE_HEADERS})
        target_include_directories(pbc_bridge PUBLIC include)
        target_link_libraries(pbc_bridge PUBLIC pbc_core Qt6::Core Qt6::Gui Qt6::Quick)
    else()
        message(STATUS "Qt6 Core/Gui/Quick not found - the Qt bridge, the app and the bridge tests are not built")
    endif()
endif()
